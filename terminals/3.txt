import serial, time, sys

def test_port(port, sends, listen_sec=4, label=''):
    print(f'\\n===== {label} ({port}) =====')
    try:
        ser = serial.Serial(port, 115200, timeout=0.3)
        time.sleep(0.5)
        ser.reset_input_buffer()
        for cmd in sends:
            line = cmd if cmd.endswith('\\n') else cmd + '\\n'
            print(f'  >> {line.strip()}')
            ser.write(line.encode())
            ser.flush()
            time.sleep(0.4)
        deadline = time.time() + listen_sec
        lines = []
        while time.time() < deadline:
            raw = ser.readline()
            if raw:
                try:
                    txt = raw.decode('utf-8', errors='replace').strip()
                except:
                    txt = repr(raw)
                if txt:
                    lines.append(txt)
                    print(f'  << {txt}')
        ser.close()
        print(f'  ({len(lines)} lines received)')
        return lines
    except Exception as e:
        print(f'  ERROR: {e}')
        return []
